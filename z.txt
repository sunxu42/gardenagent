import torch
import torch.nn as nn

class MHA(nn.Module):
    def __init__(self, d_model, num_heads):
        super(MHA, self).__init__()

        self.d_model = d_model
        self.num_heads = num_heads
        assert d_model % num_heads==0

        self.d_k = d_model // num_heads

        self.wq = nn.Linear(d_model, d_model)
        self.wk = nn.Linear(d_model, d_model)
        self.wv = nn.Linear(d_model, d_model)
        self.wo = nn.Linear(d_model, d_model)

    def forward(self, x, mask=None):
        batch_size = x.size(0)
        seq_len = x.size(1)

        q = self.wq(x).view(batch_size, -1, self.num_heads, self.d_k).transpose(1,2)
        k = self.wk(x).view(batch_size, -1, self.num_heads, self.d_k).transpose(1,2)
        v = self.wv(x).view(batch_size, -1, self.num_heads, self.d_k).transpose(1,2)

        scores = torch.matmul(q, k.transpose(2,3))/torch.sqrt(torch.tensor(self.d_k))
        if mask is not None:
            scores = scores.masked_fill(mask==0, -1e9)
        
        attn_weight = torch.nn.functional.softmax(scores, dim=-1)

        attn = torch.matmul(attn_weight, v).transpose(1,2).view(batch_size, seq_len, self.d_model)
        out = self.wo(attn)
        return out

    
class GQA(nn.Module):
    def __init__(self, d_model, num_heads, num_kv_heads):
        super(GQA, self).__init__()
    self.d_model = d_model
    self.num_heads = num_heads
    self.num_kv_heads = num_kv_heads
    assert d_model % num_heads==0
    assert num_heads % num_kv_heads==0
    
    self.dk = d_model//num_heads
    self.repeats = num_heads//num_kv_heads
    
    self.wq = nn.Linear(d_model, d_model)
    self.wk = nn.Linear(d_model, num_kv_heads*self.dk)
    self.wv = nn.Linear(d_model, num_kv_heads*self.dk)
    self.wo = nn.Linear(d_model, d_model)

def forward(self, x, mask=None):
    batch_size = x.size(0)
    seq_len = x.size(1)
    
    q = self.wq(x).view(batch_size, -1, self.num_heads, self.dk).transpose(1,2)
    k = self.wk(x).view(batch_size, -1, self.num_kv_heads, self.dk).transpose(1,2)
    v = self.wv(x).view(batch_size, -1, self.num_kv_heads, self.dk).transpose(1,2)
    
    k = k.repeat_interleave(self.repeats, dim=1)
    v = v.repeat_interleave(self.repeats, dim=1)
    
    scores = torch.matmul(q, k.transpose(2,3))/torch.sqrt(torch.tensor(self.dk, dtype=q.dtype))
    if mask is not None:
        scores = scores.masked_fill(mask==0, -1e9)
    attn_weight = torch.softmax(scores, dim=-1)
    
    attn = torch.matmul(attn_weight, v).transpose(1,2).view(batch_size, seq_len, self.d_model)
    out = self.wo(attn)
    return out
		
		

class ROPE(nn.Module):
	def __init__(self, head_dim, base=10000):
		super(ROPE, self).__init__()
		self.head_dim = head_dim
		
		inv_freq = 1.0/(base**(torch.arange(0, head_dim, 2).float()/head_dim))
		
		self.register_buffer("inv_freq", inv_freq)
		
	def forward(self, position_ids):
		# position_ids [batch_size, seq_len]
		# cos, sin [batch_size, seq_len, dim]
		freqs = torch.einsum("bi,j->bij", position_ids.float(), self.inv_freq)
		emb = torch.cat([freqs, freqs], dim=-1)
		return emb.cos(), emb.sin()

def rotate_half(x):
	# x [batch_size, num_heads, seq_len, head_dim]
	dim = x.size(-1)
	x0 = x[...,:dim]
	x1 = x[..., dim:]
	x = torch.cat([-x1, x0], dim=-1)
	return x

def apply_rotate_emb(q,k, cos, sin):
	q_emb = q*cos + rotate_half(q)*sin
	k_emb = k*cos + rotate_half(k)*sin
	return q_emb, k_emb

		emb = torch.cat([freqs, freqs], dim=-1)
		return emb.cos(), emb.sin()
		
def rotate_half(x):
	dim = x.size(-1)
	x0 = x[..., :dim//2]
	x1 = x[..., dim//2:]
	return torch.cat([-x1,x0], dim=-1)
	
def apply_rope(q,k,cos, sin):
	cos = cos.unsqueeze(1)
	sin = sin.unsqueeze(1)
	q_rot = q * cos + rotate_half(q) * sin
	k_rot = k * cos + rotate_half(k) * sin
	return q_rot, k_rot
